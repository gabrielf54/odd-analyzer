```ts
private async teamConceded(
  oppTeamId: string | null, teamField: string, window = 10,
  opts: { byPlayer?: boolean } = {},
): Promise<number | null> {
  if (!oppTeamId) return null;
  const tid = String(oppTeamId);
  let events: any;
  try { events = await this.client.events({ team_id: tid, status: "finished", limit: window }); }
  catch { return null; }
  const results: any[] = Array.isArray(events?.results) ? events.results : [];
  if (opts.byPlayer) {
    let total = 0, games = 0;
    for (const ev of results.slice(0, window)) {
      let rows: any[];
      try { rows = (await this.client.eventPlayerStats(ev.id))?.player_stats ?? []; }
      catch { continue; }
      const against = rows.filter((r) => r.player_id != null && String(r.team_id) !== tid);
      if (against.length === 0) continue;
      games++;
      for (const r of against) total += Number(r[teamField] ?? 0);
    }
    return games === 0 ? null : total / games;
  }
  const values: number[] = [];
  for (const ev of results.slice(0, window)) {
    const concededSide = String(ev.home_team_id) === tid ? "away" : "home";
    let value: any = null;
    try { value = ((await this.client.eventStats(ev.id))?.stats ?? {})[concededSide]?.[teamField]; }
    catch { value = null; }
    if (value != null) values.push(Number(value));
  }
  return values.length === 0 ? null : values.reduce((a, b) => a + b, 0) / values.length;
}
```

### Task 3: Recalibrar priors/baseline do desarme p/ won_tackle (script + medir)

**Files:**

- [ ] **Step 1: Escrever o script** (molde: `scripts/backtest-tilt.mts`). Mede, em jogos de seleção (ligas 31/58-67), com leave-one-out anti-espiada:
  - `prior_rate_won` = won_tackle/90 médio da população (jogador-jogos, min 60').
  - `baseline_won` = média entre seleções de "won_tackle concedido/jogo" (soma won_tackle dos adversários / jogos), via o mesmo padrão `byPlayer`.
  - calibração: para cada jogador-jogo, μ via `playerMuFromGames(games, "won_tackle", prior_rate_won, 12, oppMult, ha, 90)`; `prob = probOverLine(μ, 1.5, 3.0)`; `actual = won_tackle>1.5?1:0`. Imprimir `brierScore`, `expectedCalibrationError`, `brierSkillScore` (de `src/backtest/calibration.js`), e o mesmo para `total_tackle` como comparação.

```ts
// scripts/recalibrate-desarmes-won.mts — uso: npx tsx scripts/recalibrate-desarmes-won.mts
import "dotenv/config";
import { BSDClient } from "../src/bsd/client.js";
import { getRequired } from "../src/config.js";
import { playerMuFromGames } from "../src/model/assemble.js";
import { probOverLine } from "../src/model/sotModel.js";
import { opponentMultiplier, homeAwayMultiplier } from "../src/model/adjustments.js";
import { brierScore, expectedCalibrationError, brierSkillScore } from "../src/backtest/calibration.js";

const LEAGUES = [31, 58, 59, 60, 61, 62, 64, 65, 66, 67];
const EVENTS_PER_LEAGUE = 12, TARGET = 200, LINE = 1.5, R = 3.0, PRIOR_STRENGTH = 12;
const client = new BSDClient(getRequired("BZZOIRO_API_KEY"));

async function concededWonBefore(oppId: string, beforeDate: string) {
  let evs: any; try { evs = await client.events({ team_id: oppId, status: "finished", limit: 30 }); } catch { return null; }
  const past = (evs?.results ?? []).filter((e: any) => String(e.event_date ?? "") < beforeDate).slice(0, 10);
  let total = 0, games = 0;
  for (const ev of past) {
    let rows: any[]; try { rows = (await client.eventPlayerStats(ev.id))?.player_stats ?? []; } catch { continue; }
    const against = rows.filter((r) => r.player_id != null && String(r.team_id) !== oppId);
    if (!against.length) continue; games++;
    for (const r of against) total += Number(r.won_tackle ?? 0);
  }
  return games ? total / games : null;
}

// 1ª passada: estimar prior_rate_won e baseline_won (médias da população)
let sumWon90 = 0, nWon = 0, sumBaseline = 0, nBaseline = 0;
const samples: { games: any[]; oppId: string; evDate: string; isHome: boolean; actual: number }[] = [];
for (const lg of LEAGUES) {
  let evs: any; try { evs = await client.events({ league_id: lg, status: "finished", limit: EVENTS_PER_LEAGUE }); } catch { continue; }
  for (const ev of (evs?.results ?? []).slice(0, EVENTS_PER_LEAGUE)) {
    let rows: any[]; try { rows = (await client.eventPlayerStats(ev.id))?.player_stats ?? []; } catch { continue; }
    const evDate = String(ev.event_date ?? "");
    const bl = await concededWonBefore(String(ev.home_team_id), evDate); if (bl != null) { sumBaseline += bl; nBaseline++; }
    for (const r of rows) {
      if (r.player_id == null || !(Number(r.minutes_played ?? 0) >= 60)) continue;
      const won = Number(r.won_tackle ?? 0), mins = Number(r.minutes_played ?? 90);
      sumWon90 += won * 90 / mins; nWon++;
      const isHome = String(r.team_id) === String(ev.home_team_id);
      const oppId = String(isHome ? ev.away_team_id : ev.home_team_id);
      let hist: any; try { hist = await client.playerStats(r.player_id); } catch { continue; }
      const games = [...(hist?.results ?? [])].reverse().filter((g: any) => String(g.event_id) !== String(ev.id));
      if (games.length < 10) continue;
      samples.push({ games, oppId, evDate, isHome, actual: won > LINE ? 1 : 0 });
    }
    if (samples.length >= TARGET) break;
  }
  if (samples.length >= TARGET) break;
}
const priorWon = nWon ? sumWon90 / nWon : 0.8;
const baselineWon = nBaseline ? sumBaseline / nBaseline : 10.0;
console.log(`prior_rate_won≈${priorWon.toFixed(3)}  baseline_won≈${baselineWon.toFixed(2)}  (nWon=${nWon}, nBaseline=${nBaseline})`);

// 2ª passada: calibração leave-one-out com os priors estimados
const preds: number[] = [], actuals: number[] = [];
for (const s of samples) {
  const oppConc = await concededWonBefore(s.oppId, s.evDate);
  const oppMult = oppConc != null ? Math.max(0.7, Math.min(1.4, opponentMultiplier(oppConc, baselineWon))) : 1.0;
  const ha = homeAwayMultiplier(s.isHome, 1.08, 0.92);
  const calc = playerMuFromGames(s.games, "won_tackle", priorWon, PRIOR_STRENGTH, oppMult, ha, 90);
  if (!calc) continue;
  preds.push(probOverLine(calc.mu, LINE, R)); actuals.push(s.actual);
}
console.log(`won_tackle (n=${preds.length}, over ${LINE}): Brier=${brierScore(preds, actuals).toFixed(4)} ECE=${expectedCalibrationError(preds, actuals).toFixed(4)} BrierSkill=${brierSkillScore(preds, actuals).toFixed(4)}`);
```

```ts
import { brierScore, expectedCalibrationError, brierSkillScore } from "./calibration.js";
export interface MarketReport { market_type: string; n: number; brier: number; ece: number; brier_skill: number; clv_medio: number | null; }
export function reportByMarket(rows: any[]): MarketReport[] {
  const settled = rows.filter((r) => r.resultado === 0 || r.resultado === 1);
  const byMk = new Map<string, any[]>();
  for (const r of settled) { const b = byMk.get(r.market_type) ?? []; b.push(r); byMk.set(r.market_type, b); }
  const out: MarketReport[] = [];
  for (const [mk, rs] of byMk) {
    const preds = rs.map((r) => r.prob_modelo as number), acts = rs.map((r) => r.resultado as number);
    const clvs = rs.map((r) => r.clv).filter((x) => typeof x === "number");
    out.push({ market_type: mk, n: rs.length, brier: brierScore(preds, acts),
      ece: expectedCalibrationError(preds, acts), brier_skill: brierSkillScore(preds, acts),
      clv_medio: clvs.length ? clvs.reduce((a: number, b: number) => a + b, 0) / clvs.length : null });
  }
  return out;
}
```
